cmake_minimum_required(VERSION 3.16)
project(constraint_lab VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(solver src/main.cpp)
add_executable(engine_tests tests/engine_tests.cpp)
foreach(target solver engine_tests)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
enable_testing()
add_test(NAME engine COMMAND engine_tests)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME reproducible COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/run_tests.py --solver $<TARGET_FILE:solver> --report ${CMAKE_CURRENT_BINARY_DIR}/test-report.json)
endif()
